import React from 'react';

interface ProductivitySnapshotProps {
  efficiencyMetric?: number;
  weeklyDelta?: number;
  distribution?: { label: string; percentage: number }[];
}

export const ProductivitySnapshot: React.FC<ProductivitySnapshotProps> = ({ 
  efficiencyMetric = 88, 
  weeklyDelta = 4.2,
  distribution = [
    { label: 'Engineering Execution', percentage: 55 },
    { label: 'Architectural Analysis', percentage: 30 },
    { label: 'Operational Syncs', percentage: 15 }
  ]
}) => {
  return (
    <div className="p-6 rounded-xl border border-slate-200 dark:border-slate-800 bg-white/70 dark:bg-slate-900/70 backdrop-blur-md">
      <h3 className="text-lg font-bold text-slate-900 dark:text-white mb-4">Optimization Vector</h3>
      
      {/* Structural Data Layout Grid */}
      <div className="flex items-baseline gap-2">
        <span className="text-4xl font-extrabold tracking-tight text-slate-900 dark:text-white">{efficiencyMetric}%</span>
        <span className={`text-xs font-bold ${weeklyDelta >= 0 ? 'text-emerald-500' : 'text-rose-500'}`}>
          {weeklyDelta >= 0 ? '↑' : '↓'} {Math.abs(weeklyDelta)}% vs baseline
        </span>
      </div>
      <p className="text-xs text-slate-400 mt-1">Aggregate workspace performance index</p>

      {/* Direct Native Flex/Bar Layout Visualization */}
      <div className="mt-5 space-y-3">
        <div>
          <div className="flex justify-between text-2xs font-semibold uppercase tracking-wider text-slate-500 mb-1">
            <span>Resource Vector Allocation</span>
          </div>
          <div className="h-2 w-full bg-slate-100 dark:bg-slate-800 rounded-full flex overflow-hidden">
            {distribution.map((d, index) => {
              const bgColors = ['bg-primary-500', 'bg-indigo-500', 'bg-cyan-500'];
              return (
                <div 
                  key={index}
                  className={`${bgColors[index % bgColors.length]} h-full`}
                  style={{ width: `${d.percentage}%` }}
                  title={`${d.label}: ${d.percentage}%`}
                />
              );
            })}
          </div>
        </div>

        {/* Legend Mapping */}
        <div className="pt-2 space-y-1.5">
          {distribution.map((d, index) => {
            const dotColors = ['bg-primary-500', 'bg-indigo-500', 'bg-cyan-500'];
            return (
              <div key={index} className="flex items-center justify-between text-xs">
                <div className="flex items-center gap-2 text-slate-600 dark:text-slate-400 min-w-0">
                  <span className={`w-2 h-2 rounded-full flex-shrink-0 ${dotColors[index % dotColors.length]}`} />
                  <span className="truncate font-medium">{d.label}</span>
                </div>
                <span className="font-mono text-slate-500 font-bold ml-2">{d.percentage}%</span>
              </div>
            );
          })}
        </div>
      </div>
    </div>
  );
};

export default ProductivitySnapshot;
